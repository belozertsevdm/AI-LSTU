# Нечёткая экспертная система подбора сложности блюда для формирования списка покупок
# Формат:
# VAR <имя> <минимум> <максимум>
# TERM <имя_терма> Треугольник <a> <b> <c>
# ЕСЛИ Переменная=Терм И Переменная=Терм ТО Выходная_переменная=Терм

VAR Время_готовки 0 180
TERM Очень_мало Треугольник 0 0 25
TERM Мало Треугольник 10 25 45
TERM Ниже_среднего Треугольник 30 50 70
TERM Средне Треугольник 50 75 100
TERM Выше_среднего Треугольник 80 105 130
TERM Много Треугольник 110 140 165
TERM Очень_много Треугольник 150 180 180

VAR Бюджет 0 5000
TERM Низкий Треугольник 0 0 1200
TERM Ограниченный Треугольник 700 1600 2500
TERM Средний Треугольник 1800 3000 4200
TERM Высокий Треугольник 3500 5000 5000

VAR Количество_порций 1 10
TERM Мало Треугольник 1 1 3
TERM Средне Треугольник 2 5 8
TERM Много Треугольник 6 10 10

VAR Сложность_блюда 0 100
TERM Очень_простое Треугольник 0 0 20
TERM Простое Треугольник 10 25 45
TERM Среднее Треугольник 30 50 70
TERM Сложное Треугольник 55 75 90
TERM Очень_сложное Треугольник 80 100 100

# Часть A. Семь правил для одной входной переменной: Время_готовки
ЕСЛИ Время_готовки=Очень_мало ТО Сложность_блюда=Очень_простое
ЕСЛИ Время_готовки=Мало ТО Сложность_блюда=Простое
ЕСЛИ Время_готовки=Ниже_среднего ТО Сложность_блюда=Простое
ЕСЛИ Время_готовки=Средне ТО Сложность_блюда=Среднее
ЕСЛИ Время_готовки=Выше_среднего ТО Сложность_блюда=Среднее
ЕСЛИ Время_готовки=Много ТО Сложность_блюда=Сложное
ЕСЛИ Время_готовки=Очень_много ТО Сложность_блюда=Очень_сложное

# Часть B. Правила для нескольких входных переменных
ЕСЛИ Время_готовки=Очень_мало И Бюджет=Низкий И Количество_порций=Мало ТО Сложность_блюда=Очень_простое
ЕСЛИ Время_готовки=Мало И Бюджет=Низкий ТО Сложность_блюда=Очень_простое
ЕСЛИ Время_готовки=Мало И Количество_порций=Много ТО Сложность_блюда=Простое
ЕСЛИ Время_готовки=Ниже_среднего И Бюджет=Ограниченный ТО Сложность_блюда=Простое
ЕСЛИ Время_готовки=Средне И Бюджет=Низкий ТО Сложность_блюда=Простое
ЕСЛИ Время_готовки=Средне И Бюджет=Средний И Количество_порций=Средне ТО Сложность_блюда=Среднее
ЕСЛИ Время_готовки=Выше_среднего И Бюджет=Ограниченный ТО Сложность_блюда=Среднее
ЕСЛИ Время_готовки=Много И Бюджет=Низкий ТО Сложность_блюда=Среднее
ЕСЛИ Время_готовки=Много И Бюджет=Высокий И Количество_порций=Средне ТО Сложность_блюда=Сложное
ЕСЛИ Время_готовки=Очень_много И Бюджет=Высокий И Количество_порций=Много ТО Сложность_блюда=Очень_сложное
ЕСЛИ Время_готовки=Выше_среднего И Бюджет=Высокий И Количество_порций=Мало ТО Сложность_блюда=Сложное
ЕСЛИ Время_готовки=Очень_мало И Количество_порций=Много ТО Сложность_блюда=Простое
